01

Defining Structs

Structs group related data together into a named type. Each field has a name and type.

Basic Struct 
struct Point {
    x: int
    y: int
}

fn main() {
    // Create with struct literal
    let origin = Point { x: 0, y: 0 }
    let point = Point { x: 10, y: 20 }

    // Access fields with dot notation
    println(point.x)  // 10
    println(point.y)  // 20
}

main()

Fields are listed in the struct body with name: type syntax. Create instances with struct literals, providing values for all fields.

Multiple Field Types 
struct User {
    id: int
    name: string
    email: string
    active: bool
}

fn main() {
    let user = User {
        id: 1,
        name: "Alice",
        email: "alice@example.com",
        active: true
    }

    println("User: " + user.name)
    println("Email: " + user.email)
    println("Active: " + user.active)
}

main()
Nested Structs 
struct Address {
    street: string
    city: string
    zip: string
}

struct Person {
    name: string
    address: Address
}

fn main() {
    let person = Person {
        name: "Bob",
        address: Address {
            street: "123 Main St",
            city: "Springfield",
            zip: "12345"
        }
    }

    println(person.name)              // Bob
    println(person.address.city)      // Springfield
    println(person.address.street)    // 123 Main St
}

main()

Structs can contain other structs. Access nested fields with chained dot notation.

02

Impl Blocks

The impl block adds methods to a struct, separating data from behavior.

Basic Impl Block 
struct Rectangle {
    width: int
    height: int
}

impl Rectangle {
    fn area(self) -> int {
        return self.width * self.height
    }

    fn perimeter(self) -> int {
        return 2 * (self.width + self.height)
    }
}

fn main() {
    let rect = Rectangle { width: 10, height: 5 }

    println("Area: " + rect.area())        // 50
    println("Perimeter: " + rect.perimeter()) // 30
}

main()

Methods defined in impl blocks take self as the first parameter. Call them with dot notation on struct instances.

Methods Inside Struct 
// Methods can also be defined inside the struct
struct Counter {
    value: int

    fn get(self) -> int {
        return self.value
    }

    fn increment(self) -> Counter {
        return Counter { value: self.value + 1 }
    }
}

fn main() {
    let c1 = Counter { value: 0 }
    println(c1.get())  // 0

    let c2 = c1.increment()
    println(c2.get())  // 1

    let c3 = c2.increment().increment()
    println(c3.get())  // 3
}

main()

Methods can be defined directly inside the struct body for convenience. Structs are immutable, so methods return new instances rather than modifying in place.

Impl on Primitive Types 
// You can add methods to built-in types
impl int {
    fn abs(self) -> int {
        if self < 0 {
            return -self
        }
        return self
    }

    fn clamp(self, min_val: int, max_val: int) -> int {
        if self < min_val {
            return min_val
        }
        if self > max_val {
            return max_val
        }
        return self
    }

    fn is_even(self) -> bool {
        return self % 2 == 0
    }
}

fn main() {
    let n = -42
    println(n.abs())         // 42
    println(150.clamp(0, 100)) // 100
    println(7.is_even())      // false
    println(8.is_even())      // true
}

main()

Impl blocks work on any type, including primitives like int, string, and float. This extends the language without modifying core types.

03

Constructor Patterns

Use static methods (without self) as constructors to create instances with validation or default values.

Basic Constructor 
struct Circle {
    radius: float
}

impl Circle {
    // Static method - no 'self' parameter
    fn new(radius: float) -> Circle {
        return Circle { radius: radius }
    }

    fn unit() -> Circle {
        return Circle { radius: 1.0 }
    }

    fn area(self) -> float {
        return 3.14159 * self.radius * self.radius
    }
}

fn main() {
    // Call static method on the type
    let c1 = Circle.new(5.0)
    let c2 = Circle.unit()

    println(c1.area())  // ~78.54
    println(c2.area())  // ~3.14
}

main()

Static methods are called on the type (Circle.new()) rather than an instance. They're perfect for constructors that create new instances.

Constructor with Defaults 
struct Config {
    host: string
    port: int
    timeout: int
}

impl Config {
    // Default configuration
    fn default() -> Config {
        return Config {
            host: "localhost",
            port: 8080,
            timeout: 30
        }
    }

    // Builder-style methods
    fn with_host(self, host: string) -> Config {
        return Config {
            host: host,
            port: self.port,
            timeout: self.timeout
        }
    }

    fn with_port(self, port: int) -> Config {
        return Config {
            host: self.host,
            port: port,
            timeout: self.timeout
        }
    }
}

fn main() {
    // Start with defaults, customize as needed
    let config = Config.default()
        .with_host("api.example.com")
        .with_port(443)

    println(config.host)     // api.example.com
    println(config.port)     // 443
    println(config.timeout)  // 30
}

main()

Builder-style methods let you customize objects fluently. Each method returns a new instance with one field changed, keeping the struct immutable.

04

Instance Methods

Instance methods take self and operate on struct data. They're called on instances with dot notation.

Self Parameter 
struct Vector2 {
    x: float
    y: float
}

impl Vector2 {
    // 'self' refers to the instance
    fn length(self) -> float {
        return (self.x * self.x + self.y * self.y).sqrt()
    }

    fn add(self, other: Vector2) -> Vector2 {
        return Vector2 {
            x: self.x + other.x,
            y: self.y + other.y
        }
    }

    fn scale(self, factor: float) -> Vector2 {
        return Vector2 {
            x: self.x * factor,
            y: self.y * factor
        }
    }

    fn dot(self, other: Vector2) -> float {
        return self.x * other.x + self.y * other.y
    }
}

fn main() {
    let v1 = Vector2 { x: 3.0, y: 4.0 }
    let v2 = Vector2 { x: 1.0, y: 2.0 }

    println(v1.length())        // 5.0

    let sum = v1.add(v2)
    println(sum.x)              // 4.0
    println(sum.y)              // 6.0

    let scaled = v1.scale(2.0)
    println(scaled.x)           // 6.0
    println(scaled.y)           // 8.0
}

main()
Method Chaining 
struct StringBuilder {
    content: string
}

impl StringBuilder {
    fn new() -> StringBuilder {
        return StringBuilder { content: "" }
    }

    fn append(self, text: string) -> StringBuilder {
        return StringBuilder { content: self.content + text }
    }

    fn append_line(self, text: string) -> StringBuilder {
        return StringBuilder { content: self.content + text + "\n" }
    }

    fn build(self) -> string {
        return self.content
    }
}

fn main() {
    let result = StringBuilder.new()
        .append_line("# Report")
        .append_line("")
        .append("Total: ")
        .append("100")
        .build()

    println(result)
}

main()

Methods that return Self enable fluent chaining. Each method creates a new instance, keeping the original unchanged.

05

Enums

Enums define types with a fixed set of variants. Each variant can optionally carry data.

Simple Enums 
enum Color {
    Red,
    Green,
    Blue
}

fn to_hex(color: Color) -> string {
    return match color {
        Color::Red => "#FF0000",
        Color::Green => "#00FF00",
        Color::Blue => "#0000FF"
    }
}

fn main() {
    let c = Color::Red
    println(to_hex(c))  // #FF0000
}

main()

Use :: to access enum variants. Simple enums (without data) are like C-style enums but type-safe.

Enums with Data 
enum Message {
    Text(string),
    Number(int),
    Pair(int, int),
    Empty
}

fn describe(msg: Message) -> string {
    return match msg {
        Message::Text(s) => "Text: " + s,
        Message::Number(n) => "Number: " + n,
        Message::Pair(a, b) => "Pair: (" + a + ", " + b + ")",
        Message::Empty => "Empty message"
    }
}

fn main() {
    let m1 = Message::Text("Hello")
    let m2 = Message::Number(42)
    let m3 = Message::Pair(10, 20)
    let m4 = Message::Empty

    println(describe(m1))  // Text: Hello
    println(describe(m2))  // Number: 42
    println(describe(m3))  // Pair: (10, 20)
    println(describe(m4))  // Empty message
}

main()

Variants can carry different types and amounts of data. Use pattern matching to extract the data from each variant.

Generic Enums 
enum Option<T> {
    Some(T),
    None
}

enum Result<T, E> {
    Ok(T),
    Err(E)
}

fn divide(a: int, b: int) -> Result<int, string> {
    if b == 0 {
        return Result::Err("Division by zero")
    }
    return Result::Ok(a / b)
}

fn main() {
    let some_int = Option::Some(42)
    let some_str = Option::Some("hello")
    let nothing: Option<int> = Option::None

    match divide(10, 2) {
        Result::Ok(v) => println("Result: " + v),
        Result::Err(e) => println("Error: " + e)
    }
}

main()

Generic enums like Option<T> and Result<T, E> work with any types. They're fundamental patterns for handling absence and errors.

06

Composition

Build complex types by composing simpler ones. Prefer composition over inheritance.

Composing Structs 
struct Position {
    x: int
    y: int
}

struct Velocity {
    dx: int
    dy: int
}

struct Entity {
    name: string
    pos: Position
    vel: Velocity
}

impl Entity {
    fn new(name: string) -> Entity {
        return Entity {
            name: name,
            pos: Position { x: 0, y: 0 },
            vel: Velocity { dx: 0, dy: 0 }
        }
    }

    fn update(self) -> Entity {
        return Entity {
            name: self.name,
            pos: Position {
                x: self.pos.x + self.vel.dx,
                y: self.pos.y + self.vel.dy
            },
            vel: self.vel
        }
    }

    fn set_velocity(self, dx: int, dy: int) -> Entity {
        return Entity {
            name: self.name,
            pos: self.pos,
            vel: Velocity { dx: dx, dy: dy }
        }
    }
}

fn main() {
    let player = Entity.new("Player")
        .set_velocity(1, 2)
        .update()
        .update()
        .update()

    println(player.name)    // Player
    println(player.pos.x)   // 3
    println(player.pos.y)   // 6
}

main()

Compose types by including them as fields. This creates clear ownership and enables modular design - each struct focuses on one responsibility.
