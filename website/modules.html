Import Syntax

Use import statements to bring modules into scope. Imports appear at the top of your file.

Basic Import 
// Import entire module
import std.core
import std.fs
import std.json

fn main() {
    // Functions are now available
    let value = abs(-42)
    println(value)  // 42
}

main()

Importing a module makes all its public functions and types available. The standard library modules are prefixed with std.

Import Path Resolution 
// Standard library modules
import std.core      // Core functions (abs, min, max, etc.)
import std.fs        // File system operations
import std.json      // JSON parsing and formatting
import std.http      // HTTP client
import std.time      // Time and date functions
import std.math      // Mathematical functions

// Local modules (relative to current file)
import utils         // ./utils.li
import models.user   // ./models/user.li

Dots in import paths map to directory separators. std.core refers to the standard library's core module, while models.user refers to a local file.

03

Core Module

The std.core module provides essential utility functions used in most programs.

Numeric Utilities 
import std.core

fn main() {
    // Absolute value
    println(abs(-42))     // 42
    println(abs(42))      // 42

    // Minimum and maximum
    println(min(5, 3))    // 3
    println(max(5, 3))    // 5

    // Clamp to range
    println(clamp(150, 0, 100))  // 100
    println(clamp(-5, 0, 100))   // 0
    println(clamp(50, 0, 100))   // 50
}

main()
Collection Functions 
import std.core

fn main() {
    let items = [10, 20, 30, 40, 50]

    // Length
    println(len(items))  // 5

    // String length
    let text = "Hello, World!"
    println(len(text))   // 13
}

main()
Type Conversion 
import std.core

fn main() {
    // Convert to string
    let num_str = to_string(42)
    println(num_str)  // "42"

    // Parse integer from string
    let parsed = parse_int("123")
    println(parsed)  // 123

    // Parse float
    let pi = parse_float("3.14159")
    println(pi)  // 3.14159
}

File System

The std.fs module handles file and directory operations.

Reading Files 
import std.fs

fn main() {
    // Check if file exists
    if file_exists("/tmp/data.txt") {
        println("File found")
    }

    // Open file for reading (mode 0)
    let fd = file_open("/tmp/data.txt", 0)

    // Read up to 1024 bytes
    let content = file_read(fd, 1024)
    println(content)

    // Close file
    file_close(fd)
}

main()
Writing Files 
import std.fs

fn main() {
    // Open file for writing (mode 1)
    let fd = file_open("/tmp/output.txt", 1)

    // Write content
    file_write(fd, "Hello, World!\n")
    file_write(fd, "Line 2\n")
    file_write(fd, "Line 3\n")

    // Close file
    file_close(fd)

    println("File written successfully")
}

main()
File Operations 
import std.fs

fn main() {
    // Create directory
    mkdir("/tmp/myapp")

    // List directory contents
    let entries = read_dir("/tmp")
    for entry in entries {
        println(entry)
    }

    // Delete file
    remove("/tmp/old_file.txt")

    // Copy file
    copy("/tmp/source.txt", "/tmp/dest.txt")

    // Rename/move file
    rename("/tmp/old.txt", "/tmp/new.txt")
}

JSON Module

Parse and create JSON data with std.json.

Parsing JSON 
import std.json

fn main() {
    let json_str = "{\"name\": \"Alice\", \"age\": 30, \"active\": true}"

    // Parse JSON string to object
    let data = json_parse(json_str)

    // Access fields
    println(data["name"])    // Alice
    println(data["age"])     // 30
    println(data["active"])  // true
}

main()
Parsing Arrays 
import std.json

fn main() {
    let json_str = "[{\"id\": 1, \"name\": \"Alice\"}, {\"id\": 2, \"name\": \"Bob\"}]"

    let users = json_parse(json_str)

    // Iterate parsed array
    for user in users {
        println("User: " + user["name"] + " (ID: " + user["id"] + ")")
    }
}

main()
Creating JSON 
import std.json

fn main() {
    // Convert to JSON string
    let data = {
        "name": "Bob",
        "age": 25,
        "skills": ["Lira", "Rust", "Go"]
    }

    // Compact JSON string
    let compact = json_stringify(data)
    println(compact)

    // Pretty-printed JSON
    let pretty = json_pretty(data)
    println(pretty)
}

main()

json_stringify creates a compact JSON string, while json_pretty adds indentation for readability.

Organizing Code

Best practices for structuring larger Lira projects.

Project Structure 
// Recommended project layout
//
// my-project/
// ├── main.li              # Entry point
// ├── config.li            # Configuration
// ├── models/
// │   ├── user.li          # User model
// │   └── task.li          # Task model
// ├── services/
// │   ├── auth.li          # Authentication
// │   └── database.li      # Database access
// └── utils/
//     ├── validation.li    # Input validation
//     └── formatting.li    # Output formatting
Using Local Modules 
// main.li
import models.user
import models.task
import services.auth
import utils.validation

fn main() {
    // Use functions from imported modules
    let user = create_user("alice", "alice@example.com")

    if validate_email(user.email) {
        println("Valid user created")
    }
}

main()
Module Best Practices

One responsibility per module: Keep modules focused. A user.li module should only contain user-related types and functions.

Group related imports: Place standard library imports first, then local module imports, separated by a blank line.

Avoid circular dependencies: If module A imports B and B imports A, refactor shared code into a third module.

Prefer explicit exports: Only expose functions and types that other modules need. Keep implementation details private.

Example: Complete Module 
// models/user.li
import std.time

struct User {
    id: int
    name: string
    email: string
    created_at: int
}

impl User {
    fn new(id: int, name: string, email: string) -> User {
        return User {
            id: id,
            name: name,
            email: email,
            created_at: time_ms()
        }
    }

    fn display_name(self) -> string {
        return self.name + " <" + self.email + ">"
    }

    fn is_valid(self) -> bool {
        return len(self.name) > 0 && len(self.email) > 0
    }
}

// Helper function for creating users
fn create_user(name: string, email: string) -> User {
    return User.new(0, name, email)
}

A well-structured module contains related types, their methods, and helper functions. Other modules import and use these public items.
